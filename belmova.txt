Каб задаць пераменную:
зменная тэкст/літара/праўда <імя> = <значэнне>;

Друкаваць:
друкаваць(<значэнне>);

Условія:
калі (<імя> == "прывет" або <імя> == "добры дзень") тады {
	друкаваць("Добры дзень!");
}
інакш калі(<імя> == "белы" і <імя> != "чорны") тады {
	друкаваць("Сіні");
}
інакш тады{
	друкаваць("Гэй!!");	
}

цыклы:
паўтарыць (<лічба> у 3) тады {
	друкаваць(<лічба>);
}

зменная літара а = 0;
пакуль (а < 5 і а != -1) тады {
	друкаваць(а);
	а = а + 1;
}

перабраць (<элемент> у <спісак>) тады {
	друкаваць(<элемент>);
}

спісы:
зменная спіс спіс = [1, 2];   //спіс = [1, 2]
спіс.дадаць(67);              //спісак = [1, 2, 67]
друкаваць(спіс.індэкс(2));    //1
друкаваць(спіс.індэкс(67));   //2
друкаваць(спіс.індэкс(52));   //-1
друкаваць(спіс.элемент(0));   //1
друкаваць(спіс.элемент(3));   //None
спіс.выдаліць(67);            //спісак = [1, 2]
спіс.выдаліць(52);            //ПАМЫЛКА
спіс.выдаліцьІндэкс(0);       //спісак = [2]
спіс.выдаліцьІндэкс(80);      //ПАМЫЛКА
